/* Cartinha da Cacauzinha: envio por e-mail usando EmailJS */ const girlfriendMessage = $("#girlfriendMessage"); const messageFeedback = $("#messageFeedback"); $("#copyGirlfriendMessage").addEventListener("click", async () => { const message = girlfriendMessage.value.trim(); const button = $("#copyGirlfriendMessage"); if (!message) { messageFeedback.textContent = "Escreve uma cartinha primeiro, meu amor. 💗"; girlfriendMessage.focus(); return; } if (!window.emailjs) { messageFeedback.textContent = "O serviço de e-mail não carregou. Atualiza a página e tenta novamente. 💗"; return; } button.disabled = true; messageFeedback.textContent = "Enviando tua cartinha para o Ravi... 💌"; try { await window.emailjs.send( "service_ykiix53", "template_poa1fzq", { message: message } ); messageFeedback.textContent = "Cartinha enviada para o Ravi! ❤️💌"; } catch (error) { console.error("Erro ao enviar a cartinha:", error); messageFeedback.textContent = "Não foi possível enviar. Confere o EmailJS e tenta novamente. 💗"; } finally { button.disabled = false; } }); /* Botão para limpar a cartinha */ $("#clearGirlfriendMessage").addEventListener("click", () => { girlfriendMessage.value = ""; messageFeedback.textContent = "Prontinho, podes escrever outra cartinha. ♡"; girlfriendMessage.focus(); });
